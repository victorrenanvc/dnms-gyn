function near(a, b) {
  if (a === b) return true;
  if (a.length < 4 || Math.abs(a.length - b.length) > 1) return false;
  var i = 0, j = 0, d = 0;
  while (i < a.length && j < b.length) {
    if (a[i] === b[j]) { i++; j++; continue; }
    if (++d > 1) return false;
    if (a.length > b.length) i++; else if (a.length < b.length) j++; else { i++; j++; }
  }
  return d + (a.length - i) + (b.length - j) <= 1;
}

function suggestions(pool) {
  var rested = pool.filter(function (m) { return m.last && played(m).d > 14; }).sort(function (a, b) { return played(a).d - played(b).d; });
  var never = pool.filter(function (m) { return !m.last; });
  var out = rested.slice(0, 4).concat(never.slice(0, 2));
  var rest = rested.slice(4).concat(never.slice(2));
  while (out.length < 6 && rest.length) out.push(rest.shift());
  return out;
}

function toast(t) { S.toast = t; render(); setTimeout(function () { S.toast = ''; if (!quiet()) render(); }, 2200); }

function onData(d, fromCache) {
  var primeira = !S.data;
  S.data = d; S.busy = ''; S.err = ''; S.loading = false;
  if (primeira) S.anim = true;
  if (S.spin && !fromCache) { S.spin = false; S.done = true; S.anim = true; S.toast = 'Tudo atualizado'; setTimeout(function () { S.done = false; S.toast = ''; if (!quiet()) render(); }, 1600); }
  if (!fromCache) { S.offline = false; store('dnms_data', JSON.stringify(d)); }
  var ok = false;
  d.cultos.forEach(function (c) { if (c.d === S.date) ok = true; });
  if (!ok) { resetEdit(); S.date = ''; d.cultos.forEach(function (c) { if (!S.date && c.d >= d.today) S.date = c.d; }); if (!S.date && d.cultos.length) S.date = d.cultos[d.cultos.length - 1].d; }
  Object.keys(S.newSongs).forEach(function (t) { if (inRep(t)) delete S.newSongs[t]; });
  applyBridge();
  Object.keys(S.pend).forEach(function (k) { var p = S.pend[k]; d.avail = d.avail || {}; (d.avail[p.d] || (d.avail[p.d] = {}))[nrm(p.who)] = p.st; });
  refresh();
}

function onFail(e) {
  S.busy = ''; S.loading = false; S.spin = false;
  var msg = e && e.message ? e.message : String(e || '');
  if (S.data && (!navigator.onLine || /network|rede|connection|failed/i.test(msg))) S.offline = true;
  else if (!S.data) S.err = msg || 'Não consegui carregar os dados.';
  refresh();
  if (!S.data) render();
}

function verifyLeader() {
  var run = server();
  if (!run || !S.leader || !S.leader.code) return;
  var cod = S.leader.code;
  run.withSuccessHandler(function (r) { S.leader = { name: r.name, code: cod, admin: !!r.admin }; store('dnms_leader', JSON.stringify(S.leader)); refresh(); })
    .withFailureHandler(function (e) { if (/c[oó]digo/i.test(e && e.message ? e.message : '')) { ACT.logout(); refresh(); } })
    .loginLeader(cod);
}

function copy(text) {
  function fallback() {
    var ta = document.createElement('textarea'); ta.value = text; ta.style.position = 'fixed'; ta.style.opacity = '0';
    document.body.appendChild(ta); ta.focus(); ta.select();
    var ok = false; try { ok = document.execCommand('copy'); } catch (e) {}
    document.body.removeChild(ta);
    if (ok) { S.manual = ''; toast('Resumo copiado'); } else { S.manual = text; render(); }
  }
  if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(text).then(function () { S.manual = ''; toast('Resumo copiado'); }, fallback);
  else fallback();
}

/** Resposta tátil leve a cada toque. Android: vibração curta. iPhone (iOS 18+): o "tique" do sistema. */
function haptic() {
  try { if (navigator.vibrate && !/iPhone|iPad/.test(navigator.userAgent)) { navigator.vibrate(8); return; } } catch (e) {}
  try {
    var l = document.getElementById('hpt');
    if (!l) { l = document.createElement('label'); l.id = 'hpt'; l.setAttribute('aria-hidden', 'true'); l.style.cssText = 'position:fixed;left:-200px;top:0;width:1px;height:1px;overflow:hidden;opacity:0'; l.innerHTML = '<input type="checkbox" switch tabindex="-1">'; document.body.appendChild(l); }
    l.click();
  } catch (e) {}
}

  if (sub && S.me && run && (d.fresh || ((BRIDGE && BRIDGE.dnms_pushme) || store('dnms_pushme')) !== marca)) {
    if (d.fresh) S.push.busy = 'ativando lembretes…';

}
